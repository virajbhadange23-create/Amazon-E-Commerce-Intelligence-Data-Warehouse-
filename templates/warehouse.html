Warehouse DDL Architecture & Indexes

Enforcing relational integrity and index performance

-- Foreign Key Indexes for Fast Star-Join Queries
CREATE INDEX IF NOT EXISTS idx_fact_sales_product ON fact_sales (product_key);
CREATE INDEX IF NOT EXISTS idx_fact_sales_date ON fact_sales (date_key);
CREATE INDEX IF NOT EXISTS idx_fact_sales_category ON fact_sales (category_key);

-- Unique Composite Index for Strict ETL Idempotency
CREATE UNIQUE INDEX IF NOT EXISTS idx_fact_sales_unique ON fact_sales (product_key, date_key);
